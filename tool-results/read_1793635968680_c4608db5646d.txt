    12→    12→/* ---------------- noise + impulse caches ---------------- */
    13→    13→
    14→    14→function noiseBuffer(ctx: BaseAudioContext, seconds = 1): AudioBuffer {
    15→    15→  const len = Math.floor(ctx.sampleRate * seconds);
    16→    16→  const buf = ctx.createBuffer(1, len, ctx.sampleRate);
    17→    17→  const d = buf.getChannelData(0);
    18→    18→  for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
    19→    19→  return buf;
    20→    20→}
    21→    21→
    22→    22→function reverbImpulse(ctx: BaseAudioContext, seconds = 2.4, decay = 2.6): AudioBuffer {
    23→    23→  const len = Math.floor(ctx.sampleRate * seconds);
    24→    24→  const buf = ctx.createBuffer(2, len, ctx.sampleRate);
    25→    25→  for (let c = 0; c < 2; c++) {
    26→    26→    const d = buf.getChannelData(c);
    27→    27→    for (let i = 0; i < len; i++) {
    28→    28→      const t = i / len;
    29→    29→      d[i] = (Math.random() * 2 - 1) * Math.pow(1 - t, decay);
    30→    30→    }
    31→    31→  }
    32→    32→  return buf;
    33→    33→}

    48→    48→
    49→    49→export function scheduleDrum(
    50→    50→  ctx: BaseAudioContext,
    51→    51→  dest: AudioNode,
    52→    52→  sound: DrumSound,
    53→    53→  time: number,
    54→    54→  opts: DrumVoiceOpts
    55→    55→) {
    56→    56→  const { vel, tune, customBuf } = opts;
    57→    57→  if (customBuf) {
    58→    58→    const src = ctx.createBufferSource();
    59→    59→    src.buffer = customBuf;
    60→    60→    src.playbackRate.value = Math.pow(2, tune / 12);
    61→    61→    const g = ctx.createGain();
    62→    62→    g.gain.setValueAtTime(vel, time);
    63→    63→    g.gain.exponentialRampToValueAtTime(0.001, time + customBuf.duration);
    64→    64→    src.connect(g).connect(dest);
    65→    65→    src.start(time);
    66→    66→    return;
    67→    67→  }
    68→    68→  const rate = Math.pow(2, tune / 12);
    69→    69→  switch (sound) {
    70→    70→    case "kick":
    71→    71→    case "kick2": {
    72→    72→      const o = ctx.createOscillator();
    73→    73→      const g = ctx.createGain();
    74→    74→      o.type = "sine";
    75→    75→      const f0 = sound === "kick" ? 160 : 130;
    76→    76→      const f1 = sound === "kick" ? 44 : 36;
    77→    77→      const dur = sound === "kick" ? 0.42 : 0.62;
    78→    78→      o.frequency.setValueAtTime(f0 * rate, time);
    79→    79→      o.frequency.exponentialRampToValueAtTime(f1 * rate, time + 0.11);
    80→    80→      g.gain.setValueAtTime(vel, time);
    81→    81→      g.gain.exponentialRampToValueAtTime(0.001, time + dur);
    82→    82→      o.connect(g).connect(dest);
    83→    83→      o.start(time);
    84→    84→      o.stop(time + dur + 0.05);
    85→    85→      // click
    86→    86→      const c = ctx.createBufferSource();
    87→    87→      c.buffer = noiseBuffer(ctx, 0.02);
    88→    88→      const cg = ctx.createGain();
    89→    89→      cg.gain.setValueAtTime(vel * 0.35, time);
    90→    90→      cg.gain.exponentialRampToValueAtTime(0.001, time + 0.02);
    91→    91→      const cf = ctx.createBiquadFilter();
    92→    92→      cf.type = "lowpass";
    93→    93→      cf.frequency.value = 4000;
    94→    94→      c.connect(cf).connect(cg).connect(dest);
    95→    95→      c.start(time);
    96→    96→      break;
    97→    97→    }
    98→    98→    case "snare": {
    99→    99→      const n = ctx.createBufferSource();
   100→   100→      n.buffer = noiseBuffer(ctx, 0.3);
   101→   101→      const nf = ctx.createBiquadFilter();
   102→   102→      nf.type = "bandpass";
   103→   103→      nf.frequency.value = 1900 * rate;
   104→   104→      nf.Q.value = 0.9;
   105→   105→      const ng = ctx.createGain();
   106→   106→      ng.gain.setValueAtTime(vel * 0.9, time);
   107→   107→      ng.gain.exponentialRampToValueAtTime(0.001, time + 0.2);
   108→   108→      n.connect(nf).connect(ng).connect(dest);
   109→   109→      n.start(time);
   110→   110→      const o = ctx.createOscillator();
   111→   111→      o.type = "triangle";
   112→   112→      o.frequency.setValueAtTime(195 * rate, time);
   113→   113→      o.frequency.exponentialRampToValueAtTime(120 * rate, time + 0.08);
   114→   114→      const og = ctx.createGain();
   115→   115→      og.gain.setValueAtTime(vel * 0.7, time);
   116→   116→      og.gain.exponentialRampToValueAtTime(0.001, time + 0.12);
   117→   117→      o.connect(og).connect(dest);
   118→   118→      o.start(time);
   119→   119→      o.stop(time + 0.2);
   120→   120→      break;
   121→   121→    }
   122→   122→    case "clap": {
   123→   123→      for (let i = 0; i < 3; i++) {
   124→   124→        const n = ctx.createBufferSource();
   125→   125→        n.buffer = noiseBuffer(ctx, 0.15);
   126→   126→        const f = ctx.createBiquadFilter();
   127→   127→        f.type = "bandpass";
   128→   128→        f.frequency.value = 1150 * rate;
   129→   129→        f.Q.value = 1.6;
   130→   130→        const g = ctx.createGain();
   131→   131→        const t0 = time + i * 0.011;
   132→   132→        g.gain.setValueAtTime(vel * 0.8, t0);
   133→   133→        g.gain.exponentialRampToValueAtTime(0.001, t0 + (i === 2 ? 0.16 : 0.03));
   134→   134→        n.connect(f).connect(g).connect(dest);
   135→   135→        n.start(t0);
   136→   136→      }
   137→   137→      break;
   138→   138→    }
   139→   139→    case "chat":
   140→   140→    case "ohat": {
   141→   141→      const dur = sound === "chat" ? 0.05 : 0.33;
   142→   142→      const n = ctx.createBufferSource();
   143→   143→      n.buffer = noiseBuffer(ctx, 0.5);
   144→   144→      const f = ctx.createBiquadFilter();
   145→   145→      f.type = "highpass";
   146→   146→      f.frequency.value = 7200 * rate;
   147→   147→      const g = ctx.createGain();
   148→   148→      g.gain.setValueAtTime(vel * (sound === "chat" ? 0.55 : 0.5), time);
   149→   149→      g.gain.exponentialRampToValueAtTime(0.001, time + dur);
   150→   150→      n.connect(f).connect(g).connect(dest);
   151→   151→      n.start(time);
   152→   152→      break;
   153→   153→    }
   154→   154→    case "perc": {
   155→   155→      const o = ctx.createOscillator();
   156→   156→      o.type = "square";
   157→   157→      o.frequency.setValueAtTime(820 * rate, time);
   158→   158→      o.frequency.exponentialRampToValueAtTime(300 * rate, time + 0.07);
   159→   159→      const g = ctx.createGain();
   160→   160→      g.gain.setValueAtTime(vel * 0.4, time);
   161→   161→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.09);
   162→   162→      const f = ctx.createBiquadFilter();
   163→   163→      f.type = "bandpass";
   164→   164→      f.frequency.value = 900 * rate;
   165→   165→      o.connect(f).connect(g).connect(dest);
   166→   166→      o.start(time);
   167→   167→      o.stop(time + 0.15);
   168→   168→      break;
   169→   169→    }
   170→   170→    case "rim": {
   171→   171→      const o = ctx.createOscillator();
   172→   172→      o.type = "triangle";
   173→   173→      o.frequency.value = 1750 * rate;
   174→   174→      const g = ctx.createGain();
   175→   175→      g.gain.setValueAtTime(vel * 0.5, time);
   176→   176→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.045);
   177→   177→      const n = ctx.createBufferSource();
   178→   178→      n.buffer = noiseBuffer(ctx, 0.05);
   179→   179→      const nf = ctx.createBiquadFilter();
   180→   180→      nf.type = "bandpass";
   181→   181→      nf.frequency.value = 3600 * rate;
   182→   182→      const ng = ctx.createGain();
   183→   183→      ng.gain.setValueAtTime(vel * 0.4, time);
   184→   184→      ng.gain.exponentialRampToValueAtTime(0.001, time + 0.04);
   185→   185→      o.connect(g).connect(dest);
   186→   186→      n.connect(nf).connect(ng).connect(dest);
   187→   187→      o.start(time);
   188→   188→      o.stop(time + 0.1);
   189→   189→      n.start(time);
   190→   190→      break;
   191→   191→    }
   192→   192→    case "tom":
   193→   193→    case "conga": {
   194→   194→      const f0 = sound === "tom" ? 220 : 330;
   195→   195→      const o = ctx.createOscillator();
   196→   196→      o.type = "sine";
   197→   197→      o.frequency.setValueAtTime(f0 * rate, time);
   198→   198→      o.frequency.exponentialRampToValueAtTime(f0 * 0.55 * rate, time + 0.18);
   199→   199→      const g = ctx.createGain();
   200→   200→      g.gain.setValueAtTime(vel * 0.85, time);
   201→   201→      g.gain.exponentialRampToValueAtTime(0.001, time + (sound === "tom" ? 0.3 : 0.16));
   202→   202→      o.connect(g).connect(dest);
   203→   203→      o.start(time);
   204→   204→      o.stop(time + 0.4);
   205→   205→      break;
   206→   206→    }
   207→   207→    case "shaker": {
   208→   208→      const n = ctx.createBufferSource();
   209→   209→      n.buffer = noiseBuffer(ctx, 0.2);
   210→   210→      const f = ctx.createBiquadFilter();
   211→   211→      f.type = "highpass";
   212→   212→      f.frequency.value = 9000 * rate;
   213→   213→      const g = ctx.createGain();
   214→   214→      g.gain.setValueAtTime(0.001, time);
   215→   215→      g.gain.linearRampToValueAtTime(vel * 0.4, time + 0.02);
   216→   216→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.1);
   217→   217→      n.connect(f).connect(g).connect(dest);
   218→   218→      n.start(time);
   219→   219→      break;
   220→   220→    }
   221→   221→    case "ride":
   222→   222→    case "crash": {
   223→   223→      const dur = sound === "ride" ? 0.7 : 1.4;
   224→   224→      const n = ctx.createBufferSource();
   225→   225→      n.buffer = noiseBuffer(ctx, 2);
   226→   226→      const f = ctx.createBiquadFilter();
   227→   227→      f.type = "highpass";
   228→   228→      f.frequency.value = sound === "ride" ? 6800 : 4500;
   229→   229→      const g = ctx.createGain();
   230→   230→      g.gain.setValueAtTime(vel * (sound === "ride" ? 0.35 : 0.5), time);
   231→   231→      g.gain.exponentialRampToValueAtTime(0.001, time + dur);
   232→   232→      n.connect(f).connect(g).connect(dest);
   233→   233→      n.start(time);
   234→   234→      [3600, 5210, 7830].forEach((fr) => {
   235→   235→        const o = ctx.createOscillator();
   236→   236→        o.type = "square";
   237→   237→        o.frequency.value = fr * rate;
   238→   238→        const og = ctx.createGain();
   239→   239→        og.gain.setValueAtTime(vel * 0.06, time);
   240→   240→        og.gain.exponentialRampToValueAtTime(0.001, time + dur * 0.7);
   241→   241→        o.connect(og).connect(dest);
   242→   242→        o.start(time);
   243→   243→        o.stop(time + dur);
   244→   244→      });
   245→   245→      break;
   246→   246→    }
   247→   247→    case "chop": {
   248→   248→      // short filtered saw blip
   249→   249→      const o = ctx.createOscillator();
   250→   250→      o.type = "sawtooth";
   251→   251→      o.frequency.setValueAtTime(midiHz(64) * rate, time);
   252→   252→      const f = ctx.createBiquadFilter();
   253→   253→      f.type = "lowpass";
   254→   254→      f.frequency.setValueAtTime(3600 * rate, time);
   255→   255→      f.frequency.exponentialRampToValueAtTime(500, time + 0.12);
   256→   256→      f.Q.value = 6;
   257→   257→      const g = ctx.createGain();
   258→   258→      g.gain.setValueAtTime(vel * 0.5, time);
   259→   259→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.14);
   260→   260→      o.connect(f).connect(g).connect(dest);
   261→   261→      o.start(time);
   262→   262→      o.stop(time + 0.2);
   263→   263→      break;
   264→   264→    }
   265→   265→    case "vox": {
   266→   266→      // vowel-ish formant stab
   267→   267→      const o = ctx.createOscillator();
   268→   268→      o.type = "sawtooth";
   269→   269→      o.frequency.setValueAtTime(midiHz(57) * rate, time);
   270→   270→      const g = ctx.createGain();
   271→   271→      g.gain.setValueAtTime(0.001, time);
   272→   272→      g.gain.linearRampToValueAtTime(vel * 0.5, time + 0.03);
   273→   273→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.28);
   274→   274→      [[700, 1], [1220, 0.5], [2600, 0.25]].forEach(([fr, amp]) => {
   275→   275→        const f = ctx.createBiquadFilter();
   276→   276→        f.type = "bandpass";
   277→   277→        f.frequency.value = fr * rate;
   278→   278→        f.Q.value = 5;
   279→   279→        const fg = ctx.createGain();
   280→   280→        fg.gain.value = amp;
   281→   281→        o.connect(f).connect(fg).connect(g);
   282→   282→      });
   283→   283→      g.connect(dest);
   284→   284→      o.start(time);
   285→   285→      o.stop(time + 0.35);
   286→   286→      break;
   287→   287→    }
   288→   288→    case "fx": {
   289→   289→      // riser sweep
   290→   290→      const n = ctx.createBufferSource();
   291→   291→      n.buffer = noiseBuffer(ctx, 1.2);
   292→   292→      const f = ctx.createBiquadFilter();
   293→   293→      f.type = "bandpass";
   294→   294→      f.Q.value = 2.2;
   295→   295→      f.frequency.setValueAtTime(300, time);
   296→   296→      f.frequency.exponentialRampToValueAtTime(7000, time + 0.9);
   297→   297→      const g = ctx.createGain();
   298→   298→      g.gain.setValueAtTime(vel * 0.28, time);
   299→   299→      g.gain.exponentialRampToValueAtTime(0.001, time + 1.05);
   300→   300→      n.connect(f).connect(g).connect(dest);
   301→   301→      n.start(time);
   302→   302→      break;
   303→   303→    }
   304→   304→  }
   305→   305→}

   307→   307→/* ---------------- synth voice ---------------- */
   308→   308→
   309→   309→export interface SynthParams {
   310→   310→  waveform: Waveform;
   311→   311→  cutoff: number;
   312→   312→  resonance: number;
   313→   313→  attack: number;
   314→   314→  decay: number;
   315→   315→  sustain: number;
   316→   316→  release: number;
   317→   317→  detune: number;
   318→   318→  octave: number;
   319→   319→  glide: boolean;
   320→   320→}
   321→   321→
   322→   322→export function scheduleSynth(
   323→   323→  ctx: BaseAudioContext,
   324→   324→  dest: AudioNode,
   325→   325→  p: SynthParams,
   326→   326→  pitch: number,
   327→   327→  vel: number,
   328→   328→  time: number,
   329→   329→  durSec: number
   330→   330→) {
   331→   331→  const freq = midiHz(pitch + p.octave * 12);
   332→   332→  const g = ctx.createGain();
   333→   333→  const f = ctx.createBiquadFilter();
   334→   334→  f.type = "lowpass";
   335→   335→  f.Q.value = p.resonance;
   336→   336→  const cutHz = cutoffHz(p.cutoff);
   337→   337→  const peakCut = Math.min(16000, cutHz * 3.2);
   338→   338→  f.frequency.setValueAtTime(Math.max(80, peakCut), time);
   339→   339→  f.frequency.exponentialRampToValueAtTime(Math.max(90, cutHz), time + Math.max(0.03, p.decay));
   340→   340→  g.gain.setValueAtTime(0.0001, time);
   341→   341→  const peak = vel * 0.32;
   342→   342→  const atk = Math.max(0.004, p.attack);
   343→   343→  g.gain.linearRampToValueAtTime(peak, time + atk);
   344→   344→  g.gain.setTargetAtTime(peak * p.sustain, time + atk, Math.max(0.02, p.decay / 3));
   345→   345→  const end = time + Math.max(atk + 0.05, durSec);
   346→   346→  g.gain.setTargetAtTime(0.0001, end, Math.max(0.02, p.release / 3));
   347→   347→  f.connect(g).connect(dest);
   348→   348→
   349→   349→  const mkOsc = (detuneCents: number, type: OscillatorType) => {
   350→   350→    const o = ctx.createOscillator();
   351→   351→    o.type = type;
   352→   352→    if (p.glide) {
   353→   353→      o.frequency.setValueAtTime(freq * Math.pow(2, -2 / 12), time);
   354→   354→      o.frequency.exponentialRampToValueAtTime(freq, time + 0.06);
   355→   355→    } else {
   356→   356→      o.frequency.setValueAtTime(freq, time);
   357→   357→    }
   358→   358→    o.detune.value = detuneCents;
   359→   359→    o.connect(f);
   360→   360→    o.start(time);
   361→   361→    const stop = end + p.release + 0.3;
   362→   362→    o.stop(stop);
   363→   363→  };
   364→   364→  mkOsc(0, p.waveform);
   365→   365→  if (p.detune > 0) mkOsc(p.detune, p.waveform);
   366→   366→  if (p.waveform !== "sine" && p.detune > 0) mkOsc(-p.detune, p.waveform); // unison widen
   367→   367→}

   370→   370→
   371→   371→interface Channel {
   372→   372→  input: GainNode;
   373→   373→  pan: StereoPannerNode;
   374→   374→  gain: GainNode;
   375→   375→  analyser?: AnalyserNode;
   376→   376→  delaySend: GainNode;
   377→   377→  reverbSend: GainNode;
   378→   378→}
   379→   379→
   380→   380→interface Graph {
   381→   381→  master: GainNode;
   382→   382→  limiter: DynamicsCompressorNode;
   383→   383→  channels: Map<string, Channel>;
   384→   384→  delayIn: GainNode;
   385→   385→  reverbIn: GainNode;
   386→   386→  delayNode: DelayNode;
   387→   387→}
   388→   388→
   389→   389→function buildGraph(ctx: BaseAudioContext, project: Project, live: boolean): Graph {
   390→   390→  const master = ctx.createGain();
   391→   391→  master.gain.value = volGain(project.masterVolume);
   392→   392→  const limiter = ctx.createDynamicsCompressor();
   393→   393→  limiter.threshold.value = -6;
   394→   394→  limiter.knee.value = 0;
   395→   395→  limiter.ratio.value = 16;
   396→   396→  limiter.attack.value = 0.002;
   397→   397→  limiter.release.value = 0.12;
   398→   398→  master.connect(limiter);
   399→   399→
   400→   400→  let tail: AudioNode = limiter;
   401→   401→  if (live) {
   402→   402→    const analyser = ctx.createAnalyser();
   403→   403→    analyser.fftSize = 512;
   404→   404→    limiter.connect(analyser);
   405→   405→    tail = limiter;
   406→   406→    (graphExtras as { masterAnalyser?: AnalyserNode }).masterAnalyser = analyser;
   407→   407→  }
   408→   408→  tail.connect(ctx.destination);
   409→   409→
   410→   410→  // shared FX buses
   411→   411→  const spb = 60 / project.bpm;
   412→   412→  const delayIn = ctx.createGain();
   413→   413→  const delay = ctx.createDelay(2);
   414→   414→  delay.delayTime.value = spb * 0.75; // dotted 8th
   415→   415→  const fb = ctx.createGain();
   416→   416→  fb.gain.value = 0.34;
   417→   417→  const delayTone = ctx.createBiquadFilter();
   418→   418→  delayTone.type = "lowpass";
   419→   419→  delayTone.frequency.value = 3800;
   420→   420→  delayIn.connect(delay);
   421→   421→  delay.connect(delayTone);
   422→   422→  delayTone.connect(fb);
   423→   423→  fb.connect(delay);
   424→   424→  delayTone.connect(master);
   425→   425→
   426→   426→  const reverbIn = ctx.createGain();
   427→   427→  const conv = ctx.createConvolver();
   428→   428→  conv.buffer = reverbImpulse(ctx);
   429→   429→  const revTone = ctx.createBiquadFilter();
   430→   430→  revTone.type = "highpass";
   431→   431→  revTone.frequency.value = 300;
   432→   432→  reverbIn.connect(conv);
   433→   433→  conv.connect(revTone);
   434→   434→  revTone.connect(master);
   435→   435→
   436→   436→  const channels: Map<string, Channel> = new Map();
   437→   437→  const anySolo = project.tracks.some((t) => t.solo);
   438→   438→  for (const t of project.tracks) {
   439→   439→    const input = ctx.createGain();
   440→   440→    const pan = ctx.createStereoPanner();
   441→   441→    pan.pan.value = t.pan;
   442→   442→    const gain = ctx.createGain();
   443→   443→    const audible = !t.muted && (!anySolo || t.solo);
   444→   444→    gain.gain.value = audible ? volGain(t.volume) : 0;
   445→   445→    input.connect(pan);
   446→   446→    pan.connect(gain);
   447→   447→    gain.connect(master);
   448→   448→
   449→   449→    const delaySend = ctx.createGain();
   450→   450→    delaySend.gain.value = t.delaySend;
   451→   451→    gain.connect(delaySend);
   452→   452→    delaySend.connect(delayIn);
   453→   453→
   454→   454→    const reverbSend = ctx.createGain();
   455→   455→    reverbSend.gain.value = t.reverbSend;
   456→   456→    gain.connect(reverbSend);
   457→   457→    reverbSend.connect(reverbIn);
   458→   458→
   459→   459→    channels.set(t.id, { input, pan, gain, delaySend, reverbSend });
   460→   460→  }
   461→   461→  return { master, limiter, channels, delayIn, reverbIn, delayNode: delay };
   462→   462→}
   463→   463→
   464→   464→const graphExtras: { masterAnalyser?: AnalyserNode } = {};
   465→   465→
   466→   466→/* ---------------- engine ---------------- */
   467→   467→
   468→   468→export class DawEngine {
   469→   469→  private ctx: AudioContext | null = null;
   470→   470→  private graph: Graph | null = null;
   471→   471→  private timer: ReturnType<typeof setInterval> | null = null;
   472→   472→  private nextStepTime = 0;
   473→   473→  private absStep = 0;
   474→   474→  private project: Project | null = null;
   475→   475→  private queue: Array<{ step: number; time: number }> = [];
   476→   476→  private raf = 0;
   477→   477→  private noise: AudioBuffer | null = null;
   478→   478→  private customCache = new Map<string, AudioBuffer>();
   479→   479→  playing = false;
   480→   480→  onStep: ((step: number) => void) | null = null;
   481→   481→
   482→   482→  async ensure(): Promise<AudioContext> {
   483→   483→    if (!this.ctx) {
   484→   484→      const AC =
   485→   485→        window.AudioContext ||
   486→   486→        (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   487→   487→      this.ctx = new AC({ latencyHint: "interactive" });
   488→   488→      this.noise = noiseBuffer(this.ctx, 2);
   489→   489→    }
   490→   490→    if (this.ctx.state === "suspended") await this.ctx.resume();
   491→   491→    return this.ctx;
   492→   492→  }
   493→   493→
   494→   494→  async sync(project: Project) {
   495→   495→    this.project = project;
   496→   496→    const ctx = await this.ensure();
   497→   497→    if (!this.graph) {
   498→   498→      this.graph = buildGraph(ctx, project, true);
   499→   499→    } else {
   500→   500→      // update live params without rebuilding (cheap)
   501→   501→      const anySolo = project.tracks.some((t) => t.solo);
   502→   502→      this.graph.master.gain.setTargetAtTime(volGain(project.masterVolume), ctx.currentTime, 0.03);
   503→   503→      this.graph.delayNode.delayTime.setTargetAtTime((60 / project.bpm) * 0.75, ctx.currentTime, 0.08);
   504→   504→      for (const t of project.tracks) {
   505→   505→        const ch = this.graph.channels.get(t.id);
   506→   506→        if (!ch) continue;
   507→   507→        ch.pan.pan.setTargetAtTime(t.pan, ctx.currentTime, 0.03);
   508→   508→        const audible = !t.muted && (!anySolo || t.solo);
   509→   509→        ch.gain.gain.setTargetAtTime(audible ? volGain(t.volume) : 0, ctx.currentTime, 0.03);
   510→   510→        ch.delaySend.gain.setTargetAtTime(t.delaySend, ctx.currentTime, 0.03);
   511→   511→        ch.reverbSend.gain.setTargetAtTime(t.reverbSend, ctx.currentTime, 0.03);
   512→   512→      }
   513→   513→    }
   514→   514→  }
   515→   515→
   516→   516→  private rebuildIfTrackSetChanged(project: Project) {
   517→   517→    if (!this.ctx || !this.graph) return;
   518→   518→    const existing = new Set(this.graph.channels.keys());
   519→   519→    const wanted = new Set(project.tracks.map((t) => t.id));
   520→   520→    const changed =
   521→   521→      existing.size !== wanted.size || project.tracks.some((t) => !existing.has(t.id));
   522→   522→    if (changed) {
   523→   523→      this.graph = buildGraph(this.ctx, project, true);
   524→   524→    }
   525→   525→  }
   526→   526→
   527→   527→  async play(project: Project) {
   528→   528→    await this.sync(project);
   529→   529→    this.rebuildIfTrackSetChanged(project);
   530→   530→    if (!this.ctx || !this.graph) return;
   531→   531→    if (this.playing) this.stop();
   532→   532→    this.playing = true;
   533→   533→    this.absStep = 0;
   534→   534→    this.nextStepTime = this.ctx.currentTime + 0.06;
   535→   535→    this.queue = [];
   536→   536→    this.tick();
   537→   537→    this.timer = setInterval(() => this.tick(), TICK_MS);
   538→   538→    this.pump();
   539→   539→  }
   540→   540→
   541→   541→  stop() {
   542→   542→    this.playing = false;
   543→   543→    if (this.timer) clearInterval(this.timer);
   544→   544→    this.timer = null;
   545→   545→    cancelAnimationFrame(this.raf);
   546→   546→    this.queue = [];
   547→   547→    // duck the master so ringing tails cut quickly instead of smearing
   548→   548→    // over the landing page / next playback (play() restores it via sync).
   549→   549→    if (this.ctx && this.graph) {
   550→   550→      this.graph.master.gain.setTargetAtTime(0.0001, this.ctx.currentTime, 0.015);
   551→   551→    }
   552→   552→    this.onStep?.(-1);
   553→   553→  }
   554→   554→
   555→   555→  private tick() {
   556→   556→    if (!this.ctx || !this.project || !this.playing) return;
   557→   557→    const p = this.project;
   558→   558→    const spb = 60 / p.bpm;
   559→   559→    const stepDur = spb / 4;
   560→   560→    while (this.nextStepTime < this.ctx.currentTime + LOOKAHEAD_S) {
   561→   561→      const step = this.absStep % 16;
   562→   562→      const swingOffset = step % 2 === 1 ? p.swing * stepDur * 0.5 : 0;
   563→   563→      const time = this.nextStepTime + swingOffset;
   564→   564→      this.scheduleStep(p, this.graph!, step, time, stepDur);
   565→   565→      this.queue.push({ step, time });
   566→   566→      this.nextStepTime += stepDur;
   567→   567→      this.absStep++;
   568→   568→    }
   569→   569→  }
   570→   570→
   571→   571→  private scheduleStep(
   572→   572→    p: Project,
   573→   573→    graph: Graph,
   574→   574→    step: number,
   575→   575→    time: number,
   576→   576→    stepDur: number
   577→   577→  ) {
   578→   578→    const ctx = this.ctx!;
   579→   579→    const anySolo = p.tracks.some((t) => t.solo);
   580→   580→    for (const t of p.tracks) {
   581→   581→      const ch = graph.channels.get(t.id);
   582→   582→      if (!ch) continue;
   583→   583→      const audible = !t.muted && (!anySolo || t.solo);
   584→   584→      if (!audible) continue;
   585→   585→      if (t.kind === "drum") {
   586→   586→        const s = t.steps[step];
   587→   587→        if (s?.on) {
   588→   588→          scheduleDrum(ctx, ch.input, t.sound, time, {
   589→   589→            vel: s.vel,
   590→   590→            tune: t.tune,
   591→   591→            customBuf: this.customFor(t.sound),
   592→   592→          });
   593→   593→        }
   594→   594→      } else {
   595→   595→        for (const n of t.notes) {
   596→   596→          if (n.step === step) {
   597→   597→            scheduleSynth(
   598→   598→              ctx,
   599→   599→              ch.input,
   600→   600→              t as SynthTrack,
   601→   601→              n.pitch,
   602→   602→              n.vel,
   603→   603→              time,
   604→   604→              n.len * stepDur
   605→   605→            );
   606→   606→          }
   607→   607→        }
   608→   608→      }
   609→   609→    }
   610→   610→  }
   611→   611→
   612→   612→  /** fire onStep callbacks in near-realtime */
   613→   613→  private pump = () => {
   614→   614→    if (!this.playing || !this.ctx) return;
   615→   615→    const now = this.ctx.currentTime;
   616→   616→    while (this.queue.length && this.queue[0].time <= now) {
   617→   617→      const item = this.queue.shift()!;
   618→   618→      this.onStep?.(item.step);
   619→   619→    }
   620→   620→    this.raf = requestAnimationFrame(this.pump);
   621→   621→  };
   622→   622→
   623→   623→  /* ---- preview one-shots (pads, piano keys, landing) ---- */
   624→   624→
   625→   625→  async previewDrum(sound: DrumSound, vel = 0.95, tune = 0, customDataUrl?: string | null) {
   626→   626→    const ctx = await this.ensure();
   627→   627→    let buf: AudioBuffer | null = null;
   628→   628→    if (customDataUrl) buf = await this.decodeDataUrl(customDataUrl);
   629→   629→    scheduleDrum(ctx, ctx.destination, sound, ctx.currentTime + 0.01, {
   630→   630→      vel,
   631→   631→      tune,
   632→   632→      customBuf: buf,
   633→   633→    });
   634→   634→  }
   635→   635→
   636→   636→  async previewSynth(p: SynthParams, pitch: number, durSec = 0.35, vel = 0.9) {
   637→   637→    const ctx = await this.ensure();
   638→   638→    scheduleSynth(ctx, ctx.destination, p, pitch, vel, ctx.currentTime + 0.01, durSec);
   639→   639→  }
   640→   640→
   641→   641→  async decodeDataUrl(dataUrl: string): Promise<AudioBuffer | null> {
   642→   642→    if (!dataUrl) return null;
   643→   643→    const ctx = await this.ensure();
   644→   644→    const cached = this.customCache.get(dataUrl);
   645→   645→    if (cached) return cached;
   646→   646→    try {
   647→   647→      const res = await fetch(dataUrl);
   648→   648→      const arr = await res.arrayBuffer();
   649→   649→      const buf = await ctx.decodeAudioData(arr);
   650→   650→      this.customCache.set(dataUrl, buf);
   651→   651→      return buf;
   652→   652→    } catch {
   653→   653→      return null;
   654→   654→    }
   655→   655→  }
   656→   656→
   657→   657→  private customFor(sound: DrumSound): AudioBuffer | null | undefined {
   658→   658→    const pad = this.project?.pads.find((pd) => pd.sound === sound);
   659→   659→    if (pad?.custom?.dataUrl) {
   660→   660→      // async decode may not be ready; kick off and use next time
   661→   661→      void this.decodeDataUrl(pad.custom.dataUrl);
   662→   662→      return this.customCache.get(pad.custom.dataUrl) ?? null;
   663→   663→    }
   664→   664→    return null;
   665→   665→  }
   666→   666→
   667→   667→  /* ---- metering ---- */
   668→   668→
   669→   669→  private meterBuf = new Uint8Array(256);
   670→   670→
   671→   671→  level(trackId: string): number {
   672→   672→    if (!this.graph || !this.ctx) return 0;
   673→   673→    const ch = this.graph.channels.get(trackId);
   674→   674→    if (!ch) return 0;
   675→   675→    if (!ch.analyser) {
   676→   676→      const a = this.ctx.createAnalyser();
   677→   677→      a.fftSize = 256;
   678→   678→      ch.gain.connect(a);
   679→   679→      ch.analyser = a;
   680→   680→    }
   681→   681→    ch.analyser.getByteTimeDomainData(this.meterBuf);
   682→   682→    let peak = 0;
   683→   683→    for (let i = 0; i < this.meterBuf.length; i++) {
   684→   684→      const v = Math.abs(this.meterBuf[i] - 128) / 128;
   685→   685→      if (v > peak) peak = v;
   686→   686→    }
   687→   687→    return Math.min(1, peak * 1.4);
   688→   688→  }
   689→   689→
   690→   690→  masterLevel(): number {
   691→   691→    const a = graphExtras.masterAnalyser;
   692→   692→    if (!a) return 0;
   693→   693→    a.getByteTimeDomainData(this.meterBuf);
   694→   694→    let peak = 0;
   695→   695→    for (let i = 0; i < this.meterBuf.length; i++) {
   696→   696→      const v = Math.abs(this.meterBuf[i] - 128) / 128;
   697→   697→      if (v > peak) peak = v;
   698→   698→    }
   699→   699→    return Math.min(1, peak * 1.4);
   700→   700→  }

   702→   702→  /* ---- offline render → WAV ---- */
   703→   703→
   704→   704→  async renderWav(project: Project, bars: number): Promise<Blob> {
   705→   705→    const spb = 60 / project.bpm;
   706→   706→    const stepDur = spb / 4;
   707→   707→    const totalSteps = bars * 16;
   708→   708→    const dur = totalSteps * stepDur + 1.8;
   709→   709→    const off = new OfflineAudioContext(2, Math.ceil(44100 * dur), 44100);
   710→   710→    const graph = buildGraph(off, project, false);
   711→   711→    // decode user pad samples up-front so they are included in the render
   712→   712→    const padBufs = new Map<DrumSound, AudioBuffer | null>();
   713→   713→    for (const pad of project.pads) {
   714→   714→      if (pad.custom?.dataUrl && !padBufs.has(pad.sound)) {
   715→   715→        padBufs.set(pad.sound, await this.decodeDataUrl(pad.custom.dataUrl));
   716→   716→      }
   717→   717→    }
   718→   718→    const anySolo = project.tracks.some((t) => t.solo);
   719→   719→    let t = 0.05;
   720→   720→    for (let i = 0; i < totalSteps; i++) {
   721→   721→      const step = i % 16;
   722→   722→      const swingOffset = step % 2 === 1 ? project.swing * stepDur * 0.5 : 0;
   723→   723→      const time = t + swingOffset;
   724→   724→      for (const track of project.tracks) {
   725→   725→        const ch = graph.channels.get(track.id);
   726→   726→        if (!ch) continue;
   727→   727→        const audible = !track.muted && (!anySolo || track.solo);
   728→   728→        if (!audible) continue;
   729→   729→        if (track.kind === "drum") {
   730→   730→          const s = track.steps[step];
   731→   731→          if (s?.on) {
   732→   732→            scheduleDrum(off, ch.input, track.sound, time, {
   733→   733→              vel: s.vel,
   734→   734→              tune: track.tune,
   735→   735→              customBuf: padBufs.get(track.sound) ?? null,
   736→   736→            });
   737→   737→          }
   738→   738→        } else {
   739→   739→          for (const n of track.notes) {
   740→   740→            if (n.step === step) {
   741→   741→              scheduleSynth(off, ch.input, track, n.pitch, n.vel, time, n.len * stepDur);
   742→   742→            }
   743→   743→          }
   744→   744→        }
   745→   745→      }
   746→   746→      t += stepDur;
   747→   747→    }
   748→   748→    const rendered = await off.startRendering();
   749→   749→    return encodeWav(rendered);
   750→   750→  }
   751→   751→}

   753→   753→/* ---------------- WAV encoding ---------------- */
   754→   754→
   755→   755→export function encodeWav(buffer: AudioBuffer): Blob {
   756→   756→  const numCh = buffer.numberOfChannels;
   757→   757→  const len = buffer.length;
   758→   758→  const bytes = len * numCh * 2;
   759→   759→  const ab = new ArrayBuffer(44 + bytes);
   760→   760→  const view = new DataView(ab);
   761→   761→  const wstr = (off: number, s: string) => {
   762→   762→    for (let i = 0; i < s.length; i++) view.setUint8(off + i, s.charCodeAt(i));
   763→   763→  };
   764→   764→  wstr(0, "RIFF");
   765→   765→  view.setUint32(4, 36 + bytes, true);
   766→   766→  wstr(8, "WAVE");
   767→   767→  wstr(12, "fmt ");
   768→   768→  view.setUint32(16, 16, true);
   769→   769→  view.setUint16(20, 1, true); // PCM
   770→   770→  view.setUint16(22, numCh, true);
   771→   771→  view.setUint32(24, buffer.sampleRate, true);
   772→   772→  view.setUint32(28, buffer.sampleRate * numCh * 2, true);
   773→   773→  view.setUint16(32, numCh * 2, true);
   774→   774→  view.setUint16(34, 16, true);
   775→   775→  wstr(36, "data");
   776→   776→  view.setUint32(40, bytes, true);
   777→   777→
   778→   778→  const chans: Float32Array[] = [];
   779→   779→  for (let c = 0; c < numCh; c++) chans.push(buffer.getChannelData(c));
   780→   780→  let off = 44;
   781→   781→  for (let i = 0; i < len; i++) {
   782→   782→    for (let c = 0; c < numCh; c++) {
   783→   783→      let s = Math.max(-1, Math.min(1, chans[c][i]));
   784→   784→      s = s < 0 ? s * 0x8000 : s * 0x7fff;
   785→   785→      view.setInt16(off, s, true);
   786→   786→      off += 2;
   787→   787→    }
   788→   788→  }
   789→   789→  return new Blob([ab], { type: "audio/wav" });
   790→   790→}

   795→   795→  if (!_engine) _engine = new DawEngine();
   796→   796→  return _engine;
