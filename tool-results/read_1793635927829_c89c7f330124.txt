    12→/* ---------------- noise + impulse caches ---------------- */
    13→
    14→function noiseBuffer(ctx: BaseAudioContext, seconds = 1): AudioBuffer {
    15→  const len = Math.floor(ctx.sampleRate * seconds);
    16→  const buf = ctx.createBuffer(1, len, ctx.sampleRate);
    17→  const d = buf.getChannelData(0);
    18→  for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
    19→  return buf;
    20→}
    21→
    22→function reverbImpulse(ctx: BaseAudioContext, seconds = 2.4, decay = 2.6): AudioBuffer {
    23→  const len = Math.floor(ctx.sampleRate * seconds);
    24→  const buf = ctx.createBuffer(2, len, ctx.sampleRate);
    25→  for (let c = 0; c < 2; c++) {
    26→    const d = buf.getChannelData(c);
    27→    for (let i = 0; i < len; i++) {
    28→      const t = i / len;
    29→      d[i] = (Math.random() * 2 - 1) * Math.pow(1 - t, decay);
    30→    }
    31→  }
    32→  return buf;
    33→}

    49→export function scheduleDrum(
    50→  ctx: BaseAudioContext,
    51→  dest: AudioNode,
    52→  sound: DrumSound,
    53→  time: number,
    54→  opts: DrumVoiceOpts
    55→) {
    56→  const { vel, tune, customBuf } = opts;
    57→  if (customBuf) {
    58→    const src = ctx.createBufferSource();
    59→    src.buffer = customBuf;
    60→    src.playbackRate.value = Math.pow(2, tune / 12);
    61→    const g = ctx.createGain();
    62→    g.gain.setValueAtTime(vel, time);
    63→    g.gain.exponentialRampToValueAtTime(0.001, time + customBuf.duration);
    64→    src.connect(g).connect(dest);
    65→    src.start(time);
    66→    return;
    67→  }
    68→  const rate = Math.pow(2, tune / 12);
    69→  switch (sound) {
    70→    case "kick":
    71→    case "kick2": {
    72→      const o = ctx.createOscillator();
    73→      const g = ctx.createGain();
    74→      o.type = "sine";
    75→      const f0 = sound === "kick" ? 160 : 130;
    76→      const f1 = sound === "kick" ? 44 : 36;
    77→      const dur = sound === "kick" ? 0.42 : 0.62;
    78→      o.frequency.setValueAtTime(f0 * rate, time);
    79→      o.frequency.exponentialRampToValueAtTime(f1 * rate, time + 0.11);
    80→      g.gain.setValueAtTime(vel, time);
    81→      g.gain.exponentialRampToValueAtTime(0.001, time + dur);
    82→      o.connect(g).connect(dest);
    83→      o.start(time);
    84→      o.stop(time + dur + 0.05);
    85→      // click
    86→      const c = ctx.createBufferSource();
    87→      c.buffer = noiseBuffer(ctx, 0.02);
    88→      const cg = ctx.createGain();
    89→      cg.gain.setValueAtTime(vel * 0.35, time);
    90→      cg.gain.exponentialRampToValueAtTime(0.001, time + 0.02);
    91→      const cf = ctx.createBiquadFilter();
    92→      cf.type = "lowpass";
    93→      cf.frequency.value = 4000;
    94→      c.connect(cf).connect(cg).connect(dest);
    95→      c.start(time);
    96→      break;
    97→    }
    98→    case "snare": {
    99→      const n = ctx.createBufferSource();
   100→      n.buffer = noiseBuffer(ctx, 0.3);
   101→      const nf = ctx.createBiquadFilter();
   102→      nf.type = "bandpass";
   103→      nf.frequency.value = 1900 * rate;
   104→      nf.Q.value = 0.9;
   105→      const ng = ctx.createGain();
   106→      ng.gain.setValueAtTime(vel * 0.9, time);
   107→      ng.gain.exponentialRampToValueAtTime(0.001, time + 0.2);
   108→      n.connect(nf).connect(ng).connect(dest);
   109→      n.start(time);
   110→      const o = ctx.createOscillator();
   111→      o.type = "triangle";
   112→      o.frequency.setValueAtTime(195 * rate, time);
   113→      o.frequency.exponentialRampToValueAtTime(120 * rate, time + 0.08);
   114→      const og = ctx.createGain();
   115→      og.gain.setValueAtTime(vel * 0.7, time);
   116→      og.gain.exponentialRampToValueAtTime(0.001, time + 0.12);
   117→      o.connect(og).connect(dest);
   118→      o.start(time);
   119→      o.stop(time + 0.2);
   120→      break;
   121→    }
   122→    case "clap": {
   123→      for (let i = 0; i < 3; i++) {
   124→        const n = ctx.createBufferSource();
   125→        n.buffer = noiseBuffer(ctx, 0.15);
   126→        const f = ctx.createBiquadFilter();
   127→        f.type = "bandpass";
   128→        f.frequency.value = 1150 * rate;
   129→        f.Q.value = 1.6;
   130→        const g = ctx.createGain();
   131→        const t0 = time + i * 0.011;
   132→        g.gain.setValueAtTime(vel * 0.8, t0);
   133→        g.gain.exponentialRampToValueAtTime(0.001, t0 + (i === 2 ? 0.16 : 0.03));
   134→        n.connect(f).connect(g).connect(dest);
   135→        n.start(t0);
   136→      }
   137→      break;
   138→    }
   139→    case "chat":
   140→    case "ohat": {
   141→      const dur = sound === "chat" ? 0.05 : 0.33;
   142→      const n = ctx.createBufferSource();
   143→      n.buffer = noiseBuffer(ctx, 0.5);
   144→      const f = ctx.createBiquadFilter();
   145→      f.type = "highpass";
   146→      f.frequency.value = 7200 * rate;
   147→      const g = ctx.createGain();
   148→      g.gain.setValueAtTime(vel * (sound === "chat" ? 0.55 : 0.5), time);
   149→      g.gain.exponentialRampToValueAtTime(0.001, time + dur);
   150→      n.connect(f).connect(g).connect(dest);
   151→      n.start(time);
   152→      break;
   153→    }
   154→    case "perc": {
   155→      const o = ctx.createOscillator();
   156→      o.type = "square";
   157→      o.frequency.setValueAtTime(820 * rate, time);
   158→      o.frequency.exponentialRampToValueAtTime(300 * rate, time + 0.07);
   159→      const g = ctx.createGain();
   160→      g.gain.setValueAtTime(vel * 0.4, time);
   161→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.09);
   162→      const f = ctx.createBiquadFilter();
   163→      f.type = "bandpass";
   164→      f.frequency.value = 900 * rate;
   165→      o.connect(f).connect(g).connect(dest);
   166→      o.start(time);
   167→      o.stop(time + 0.15);
   168→      break;
   169→    }
   170→    case "rim": {
   171→      const o = ctx.createOscillator();
   172→      o.type = "triangle";
   173→      o.frequency.value = 1750 * rate;
   174→      const g = ctx.createGain();
   175→      g.gain.setValueAtTime(vel * 0.5, time);
   176→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.045);
   177→      const n = ctx.createBufferSource();
   178→      n.buffer = noiseBuffer(ctx, 0.05);
   179→      const nf = ctx.createBiquadFilter();
   180→      nf.type = "bandpass";
   181→      nf.frequency.value = 3600 * rate;
   182→      const ng = ctx.createGain();
   183→      ng.gain.setValueAtTime(vel * 0.4, time);
   184→      ng.gain.exponentialRampToValueAtTime(0.001, time + 0.04);
   185→      o.connect(g).connect(dest);
   186→      n.connect(nf).connect(ng).connect(dest);
   187→      o.start(time);
   188→      o.stop(time + 0.1);
   189→      n.start(time);
   190→      break;
   191→    }
   192→    case "tom":
   193→    case "conga": {
   194→      const f0 = sound === "tom" ? 220 : 330;
   195→      const o = ctx.createOscillator();
   196→      o.type = "sine";
   197→      o.frequency.setValueAtTime(f0 * rate, time);
   198→      o.frequency.exponentialRampToValueAtTime(f0 * 0.55 * rate, time + 0.18);
   199→      const g = ctx.createGain();
   200→      g.gain.setValueAtTime(vel * 0.85, time);
   201→      g.gain.exponentialRampToValueAtTime(0.001, time + (sound === "tom" ? 0.3 : 0.16));
   202→      o.connect(g).connect(dest);
   203→      o.start(time);
   204→      o.stop(time + 0.4);
   205→      break;
   206→    }
   207→    case "shaker": {
   208→      const n = ctx.createBufferSource();
   209→      n.buffer = noiseBuffer(ctx, 0.2);
   210→      const f = ctx.createBiquadFilter();
   211→      f.type = "highpass";
   212→      f.frequency.value = 9000 * rate;
   213→      const g = ctx.createGain();
   214→      g.gain.setValueAtTime(0.001, time);
   215→      g.gain.linearRampToValueAtTime(vel * 0.4, time + 0.02);
   216→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.1);
   217→      n.connect(f).connect(g).connect(dest);
   218→      n.start(time);
   219→      break;
   220→    }
   221→    case "ride":
   222→    case "crash": {
   223→      const dur = sound === "ride" ? 0.7 : 1.4;
   224→      const n = ctx.createBufferSource();
   225→      n.buffer = noiseBuffer(ctx, 2);
   226→      const f = ctx.createBiquadFilter();
   227→      f.type = "highpass";
   228→      f.frequency.value = sound === "ride" ? 6800 : 4500;
   229→      const g = ctx.createGain();
   230→      g.gain.setValueAtTime(vel * (sound === "ride" ? 0.35 : 0.5), time);
   231→      g.gain.exponentialRampToValueAtTime(0.001, time + dur);
   232→      n.connect(f).connect(g).connect(dest);
   233→      n.start(time);
   234→      [3600, 5210, 7830].forEach((fr) => {
   235→        const o = ctx.createOscillator();
   236→        o.type = "square";
   237→        o.frequency.value = fr * rate;
   238→        const og = ctx.createGain();
   239→        og.gain.setValueAtTime(vel * 0.06, time);
   240→        og.gain.exponentialRampToValueAtTime(0.001, time + dur * 0.7);
   241→        o.connect(og).connect(dest);
   242→        o.start(time);
   243→        o.stop(time + dur);
   244→      });
   245→      break;
   246→    }
   247→    case "chop": {
   248→      // short filtered saw blip
   249→      const o = ctx.createOscillator();
   250→      o.type = "sawtooth";
   251→      o.frequency.setValueAtTime(midiHz(64) * rate, time);
   252→      const f = ctx.createBiquadFilter();
   253→      f.type = "lowpass";
   254→      f.frequency.setValueAtTime(3600 * rate, time);
   255→      f.frequency.exponentialRampToValueAtTime(500, time + 0.12);
   256→      f.Q.value = 6;
   257→      const g = ctx.createGain();
   258→      g.gain.setValueAtTime(vel * 0.5, time);
   259→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.14);
   260→      o.connect(f).connect(g).connect(dest);
   261→      o.start(time);
   262→      o.stop(time + 0.2);
   263→      break;
   264→    }
   265→    case "vox": {
   266→      // vowel-ish formant stab
   267→      const o = ctx.createOscillator();
   268→      o.type = "sawtooth";
   269→      o.frequency.setValueAtTime(midiHz(57) * rate, time);
   270→      const g = ctx.createGain();
   271→      g.gain.setValueAtTime(0.001, time);
   272→      g.gain.linearRampToValueAtTime(vel * 0.5, time + 0.03);
   273→      g.gain.exponentialRampToValueAtTime(0.001, time + 0.28);
   274→      [[700, 1], [1220, 0.5], [2600, 0.25]].forEach(([fr, amp]) => {
   275→        const f = ctx.createBiquadFilter();
   276→        f.type = "bandpass";
   277→        f.frequency.value = fr * rate;
   278→        f.Q.value = 5;
   279→        const fg = ctx.createGain();
   280→        fg.gain.value = amp;
   281→        o.connect(f).connect(fg).connect(g);
   282→      });
   283→      g.connect(dest);
   284→      o.start(time);
   285→      o.stop(time + 0.35);
   286→      break;
   287→    }
   288→    case "fx": {
   289→      // riser sweep
   290→      const n = ctx.createBufferSource();
   291→      n.buffer = noiseBuffer(ctx, 1.2);
   292→      const f = ctx.createBiquadFilter();
   293→      f.type = "bandpass";
   294→      f.Q.value = 2.2;
   295→      f.frequency.setValueAtTime(300, time);
   296→      f.frequency.exponentialRampToValueAtTime(7000, time + 0.9);
   297→      const g = ctx.createGain();
   298→      g.gain.setValueAtTime(vel * 0.28, time);
   299→      g.gain.exponentialRampToValueAtTime(0.001, time + 1.05);
   300→      n.connect(f).connect(g).connect(dest);
   301→      n.start(time);
   302→      break;
   303→    }
   304→  }
   305→}

   309→export interface SynthParams {
   310→  waveform: Waveform;
   311→  cutoff: number;
   312→  resonance: number;
   313→  attack: number;
   314→  decay: number;
   315→  sustain: number;
   316→  release: number;
   317→  detune: number;
   318→  octave: number;
   319→  glide: boolean;
   320→}
   321→
   322→export function scheduleSynth(
   323→  ctx: BaseAudioContext,
   324→  dest: AudioNode,
   325→  p: SynthParams,
   326→  pitch: number,
   327→  vel: number,
   328→  time: number,
   329→  durSec: number
   330→) {
   331→  const freq = midiHz(pitch + p.octave * 12);
   332→  const g = ctx.createGain();
   333→  const f = ctx.createBiquadFilter();
   334→  f.type = "lowpass";
   335→  f.Q.value = p.resonance;
   336→  const cutHz = cutoffHz(p.cutoff);
   337→  const peakCut = Math.min(16000, cutHz * 3.2);
   338→  f.frequency.setValueAtTime(Math.max(80, peakCut), time);
   339→  f.frequency.exponentialRampToValueAtTime(Math.max(90, cutHz), time + Math.max(0.03, p.decay));
   340→  g.gain.setValueAtTime(0.0001, time);
   341→  const peak = vel * 0.32;
   342→  const atk = Math.max(0.004, p.attack);
   343→  g.gain.linearRampToValueAtTime(peak, time + atk);
   344→  g.gain.setTargetAtTime(peak * p.sustain, time + atk, Math.max(0.02, p.decay / 3));
   345→  const end = time + Math.max(atk + 0.05, durSec);
   346→  g.gain.setTargetAtTime(0.0001, end, Math.max(0.02, p.release / 3));
   347→  f.connect(g).connect(dest);
   348→
   349→  const mkOsc = (detuneCents: number, type: OscillatorType) => {
   350→    const o = ctx.createOscillator();
   351→    o.type = type;
   352→    if (p.glide) {
   353→      o.frequency.setValueAtTime(freq * Math.pow(2, -2 / 12), time);
   354→      o.frequency.exponentialRampToValueAtTime(freq, time + 0.06);
   355→    } else {
   356→      o.frequency.setValueAtTime(freq, time);
   357→    }
   358→    o.detune.value = detuneCents;
   359→    o.connect(f);
   360→    o.start(time);
   361→    const stop = end + p.release + 0.3;
   362→    o.stop(stop);
   363→  };
   364→  mkOsc(0, p.waveform);
   365→  if (p.detune > 0) mkOsc(p.detune, p.waveform);
   366→  if (p.waveform !== "sine" && p.detune > 0) mkOsc(-p.detune, p.waveform); // unison widen
   367→}

   369→/* ---------------- graph ---------------- */
   370→
   371→interface Channel {
   372→  input: GainNode;
   373→  pan: StereoPannerNode;
   374→  gain: GainNode;
   375→  analyser?: AnalyserNode;
   376→  delaySend: GainNode;
   377→  reverbSend: GainNode;
   378→}
   379→
   380→interface Graph {
   381→  master: GainNode;
   382→  limiter: DynamicsCompressorNode;
   383→  channels: Map<string, Channel>;
   384→  delayIn: GainNode;
   385→  reverbIn: GainNode;
   386→  delayNode: DelayNode;
   387→}
   388→
   389→function buildGraph(ctx: BaseAudioContext, project: Project, live: boolean): Graph {
   390→  const master = ctx.createGain();
   391→  master.gain.value = volGain(project.masterVolume);
   392→  const limiter = ctx.createDynamicsCompressor();
   393→  limiter.threshold.value = -6;
   394→  limiter.knee.value = 0;
   395→  limiter.ratio.value = 16;
   396→  limiter.attack.value = 0.002;
   397→  limiter.release.value = 0.12;
   398→  master.connect(limiter);
   399→
   400→  let tail: AudioNode = limiter;
   401→  if (live) {
   402→    const analyser = ctx.createAnalyser();
   403→    analyser.fftSize = 512;
   404→    limiter.connect(analyser);
   405→    tail = limiter;
   406→    (graphExtras as { masterAnalyser?: AnalyserNode }).masterAnalyser = analyser;
   407→  }
   408→  tail.connect(ctx.destination);
   409→
   410→  // shared FX buses
   411→  const spb = 60 / project.bpm;
   412→  const delayIn = ctx.createGain();
   413→  const delay = ctx.createDelay(2);
   414→  delay.delayTime.value = spb * 0.75; // dotted 8th
   415→  const fb = ctx.createGain();
   416→  fb.gain.value = 0.34;
   417→  const delayTone = ctx.createBiquadFilter();
   418→  delayTone.type = "lowpass";
   419→  delayTone.frequency.value = 3800;
   420→  delayIn.connect(delay);
   421→  delay.connect(delayTone);
   422→  delayTone.connect(fb);
   423→  fb.connect(delay);
   424→  delayTone.connect(master);
   425→
   426→  const reverbIn = ctx.createGain();
   427→  const conv = ctx.createConvolver();
   428→  conv.buffer = reverbImpulse(ctx);
   429→  const revTone = ctx.createBiquadFilter();
   430→  revTone.type = "highpass";
   431→  revTone.frequency.value = 300;
   432→  reverbIn.connect(conv);
   433→  conv.connect(revTone);
   434→  revTone.connect(master);
   435→
   436→  const channels: Map<string, Channel> = new Map();
   437→  const anySolo = project.tracks.some((t) => t.solo);
   438→  for (const t of project.tracks) {
   439→    const input = ctx.createGain();
   440→    const pan = ctx.createStereoPanner();
   441→    pan.pan.value = t.pan;
   442→    const gain = ctx.createGain();
   443→    const audible = !t.muted && (!anySolo || t.solo);
   444→    gain.gain.value = audible ? volGain(t.volume) : 0;
   445→    input.connect(pan);
   446→    pan.connect(gain);
   447→    gain.connect(master);
   448→
   449→    const delaySend = ctx.createGain();
   450→    delaySend.gain.value = t.delaySend;
   451→    gain.connect(delaySend);
   452→    delaySend.connect(delayIn);
   453→
   454→    const reverbSend = ctx.createGain();
   455→    reverbSend.gain.value = t.reverbSend;
   456→    gain.connect(reverbSend);
   457→    reverbSend.connect(reverbIn);
   458→
   459→    channels.set(t.id, { input, pan, gain, delaySend, reverbSend });
   460→  }
   461→  return { master, limiter, channels, delayIn, reverbIn, delayNode: delay };
   462→}

   466→/* ---------------- engine ---------------- */
   467→
   468→export class DawEngine {
   469→  private ctx: AudioContext | null = null;
   470→  private graph: Graph | null = null;
   471→  private timer: ReturnType<typeof setInterval> | null = null;
   472→  private nextStepTime = 0;
   473→  private absStep = 0;
   474→  private project: Project | null = null;
   475→  private queue: Array<{ step: number; time: number }> = [];
   476→  private raf = 0;
   477→  private noise: AudioBuffer | null = null;
   478→  private customCache = new Map<string, AudioBuffer>();
   479→  playing = false;
   480→  onStep: ((step: number) => void) | null = null;
   481→
   482→  async ensure(): Promise<AudioContext> {
   483→    if (!this.ctx) {
   484→      const AC =
   485→        window.AudioContext ||
   486→        (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   487→      this.ctx = new AC({ latencyHint: "interactive" });
   488→      this.noise = noiseBuffer(this.ctx, 2);
   489→    }
   490→    if (this.ctx.state === "suspended") await this.ctx.resume();
   491→    return this.ctx;
   492→  }
   493→
   494→  async sync(project: Project) {
   495→    this.project = project;
   496→    const ctx = await this.ensure();
   497→    if (!this.graph) {
   498→      this.graph = buildGraph(ctx, project, true);
   499→    } else {
   500→      // update live params without rebuilding (cheap)
   501→      const anySolo = project.tracks.some((t) => t.solo);
   502→      this.graph.master.gain.setTargetAtTime(volGain(project.masterVolume), ctx.currentTime, 0.03);
   503→      this.graph.delayNode.delayTime.setTargetAtTime((60 / project.bpm) * 0.75, ctx.currentTime, 0.08);
   504→      for (const t of project.tracks) {
   505→        const ch = this.graph.channels.get(t.id);
   506→        if (!ch) continue;
   507→        ch.pan.pan.setTargetAtTime(t.pan, ctx.currentTime, 0.03);
   508→        const audible = !t.muted && (!anySolo || t.solo);
   509→        ch.gain.gain.setTargetAtTime(audible ? volGain(t.volume) : 0, ctx.currentTime, 0.03);
   510→        ch.delaySend.gain.setTargetAtTime(t.delaySend, ctx.currentTime, 0.03);
   511→        ch.reverbSend.gain.setTargetAtTime(t.reverbSend, ctx.currentTime, 0.03);
   512→      }
   513→    }
   514→  }
   515→
   516→  private rebuildIfTrackSetChanged(project: Project) {
   517→    if (!this.ctx || !this.graph) return;
   518→    const existing = new Set(this.graph.channels.keys());
   519→    const wanted = new Set(project.tracks.map((t) => t.id));
   520→    const changed =
   521→      existing.size !== wanted.size || project.tracks.some((t) => !existing.has(t.id));
   522→    if (changed) {
   523→      this.graph = buildGraph(this.ctx, project, true);
   524→    }
   525→  }
   526→
   527→  async play(project: Project) {
   528→    await this.sync(project);
   529→    this.rebuildIfTrackSetChanged(project);
   530→    if (!this.ctx || !this.graph) return;
   531→    if (this.playing) this.stop();
   532→    this.playing = true;
   533→    this.absStep = 0;
   534→    this.nextStepTime = this.ctx.currentTime + 0.06;
   535→    this.queue = [];
   536→    this.tick();
   537→    this.timer = setInterval(() => this.tick(), TICK_MS);
   538→    this.pump();
   539→  }
   540→
   541→  stop() {
   542→    this.playing = false;
   543→    if (this.timer) clearInterval(this.timer);
   544→    this.timer = null;
   545→    cancelAnimationFrame(this.raf);
   546→    this.queue = [];
   547→    // duck the master so ringing tails cut quickly instead of smearing
   548→    // over the landing page / next playback (play() restores it via sync).
   549→    if (this.ctx && this.graph) {
   550→      this.graph.master.gain.setTargetAtTime(0.0001, this.ctx.currentTime, 0.015);
   551→    }
   552→    this.onStep?.(-1);
   553→  }
   554→
   555→  private tick() {
   556→    if (!this.ctx || !this.project || !this.playing) return;
   557→    const p = this.project;
   558→    const spb = 60 / p.bpm;
   559→    const stepDur = spb / 4;
   560→    while (this.nextStepTime < this.ctx.currentTime + LOOKAHEAD_S) {
   561→      const step = this.absStep % 16;
   562→      const swingOffset = step % 2 === 1 ? p.swing * stepDur * 0.5 : 0;
   563→      const time = this.nextStepTime + swingOffset;
   564→      this.scheduleStep(p, this.graph!, step, time, stepDur);
   565→      this.queue.push({ step, time });
   566→      this.nextStepTime += stepDur;
   567→      this.absStep++;
   568→    }
   569→  }
   570→
   571→  private scheduleStep(
   572→    p: Project,
   573→    graph: Graph,
   574→    step: number,
   575→    time: number,
   576→    stepDur: number
   577→  ) {
   578→    const ctx = this.ctx!;
   579→    const anySolo = p.tracks.some((t) => t.solo);
   580→    for (const t of p.tracks) {
   581→      const ch = graph.channels.get(t.id);
   582→      if (!ch) continue;
   583→      const audible = !t.muted && (!anySolo || t.solo);
   584→      if (!audible) continue;
   585→      if (t.kind === "drum") {
   586→        const s = t.steps[step];
   587→        if (s?.on) {
   588→          scheduleDrum(ctx, ch.input, t.sound, time, {
   589→            vel: s.vel,
   590→            tune: t.tune,
   591→            customBuf: this.customFor(t.sound),
   592→          });
   593→        }
   594→      } else {
   595→        for (const n of t.notes) {
   596→          if (n.step === step) {
   597→            scheduleSynth(
   598→              ctx,
   599→              ch.input,
   600→              t as SynthTrack,
   601→              n.pitch,
   602→              n.vel,
   603→              time,
   604→              n.len * stepDur
   605→            );
   606→          }
   607→        }
   608→      }
   609→    }
   610→  }
   611→
   612→  /** fire onStep callbacks in near-realtime */
   613→  private pump = () => {
   614→    if (!this.playing || !this.ctx) return;
   615→    const now = this.ctx.currentTime;
   616→    while (this.queue.length && this.queue[0].time <= now) {
   617→      const item = this.queue.shift()!;
   618→      this.onStep?.(item.step);
   619→    }
   620→    this.raf = requestAnimationFrame(this.pump);
   621→  };
   622→
   623→  /* ---- preview one-shots (pads, piano keys, landing) ---- */
   624→
   625→  async previewDrum(sound: DrumSound, vel = 0.95, tune = 0, customDataUrl?: string | null) {
   626→    const ctx = await this.ensure();
   627→    let buf: AudioBuffer | null = null;
   628→    if (customDataUrl) buf = await this.decodeDataUrl(customDataUrl);
   629→    scheduleDrum(ctx, ctx.destination, sound, ctx.currentTime + 0.01, {
   630→      vel,
   631→      tune,
   632→      customBuf: buf,
   633→    });
   634→  }
   635→
   636→  async previewSynth(p: SynthParams, pitch: number, durSec = 0.35, vel = 0.9) {
   637→    const ctx = await this.ensure();
   638→    scheduleSynth(ctx, ctx.destination, p, pitch, vel, ctx.currentTime + 0.01, durSec);
   639→  }
   640→
   641→  async decodeDataUrl(dataUrl: string): Promise<AudioBuffer | null> {
   642→    if (!dataUrl) return null;
   643→    const ctx = await this.ensure();
   644→    const cached = this.customCache.get(dataUrl);
   645→    if (cached) return cached;
   646→    try {
   647→      const res = await fetch(dataUrl);
   648→      const arr = await res.arrayBuffer();
   649→      const buf = await ctx.decodeAudioData(arr);
   650→      this.customCache.set(dataUrl, buf);
   651→      return buf;
   652→    } catch {
   653→      return null;
   654→    }
   655→  }
   656→
   657→  private customFor(sound: DrumSound): AudioBuffer | null | undefined {
   658→    const pad = this.project?.pads.find((pd) => pd.sound === sound);
   659→    if (pad?.custom?.dataUrl) {
   660→      // async decode may not be ready; kick off and use next time
   661→      void this.decodeDataUrl(pad.custom.dataUrl);
   662→      return this.customCache.get(pad.custom.dataUrl) ?? null;
   663→    }
   664→    return null;
   665→  }

   667→  /* ---- metering ---- */
   668→
   669→  private meterBuf = new Uint8Array(256);
   670→
   671→  level(trackId: string): number {
   672→    if (!this.graph || !this.ctx) return 0;
   673→    const ch = this.graph.channels.get(trackId);
   674→    if (!ch) return 0;
   675→    if (!ch.analyser) {
   676→      const a = this.ctx.createAnalyser();
   677→      a.fftSize = 256;
   678→      ch.gain.connect(a);
   679→      ch.analyser = a;
   680→    }
   681→    ch.analyser.getByteTimeDomainData(this.meterBuf);
   682→    let peak = 0;
   683→    for (let i = 0; i < this.meterBuf.length; i++) {
   684→      const v = Math.abs(this.meterBuf[i] - 128) / 128;
   685→      if (v > peak) peak = v;
   686→    }
   687→    return Math.min(1, peak * 1.4);
   688→  }
   689→
   690→  masterLevel(): number {
   691→    const a = graphExtras.masterAnalyser;
   692→    if (!a) return 0;
   693→    a.getByteTimeDomainData(this.meterBuf);
   694→    let peak = 0;
   695→    for (let i = 0; i < this.meterBuf.length; i++) {
   696→      const v = Math.abs(this.meterBuf[i] - 128) / 128;
   697→      if (v > peak) peak = v;
   698→    }
   699→    return Math.min(1, peak * 1.4);
   700→  }

   702→  /* ---- offline render → WAV ---- */
   703→
   704→  async renderWav(project: Project, bars: number): Promise<Blob> {
   705→    const spb = 60 / project.bpm;
   706→    const stepDur = spb / 4;
   707→    const totalSteps = bars * 16;
   708→    const dur = totalSteps * stepDur + 1.8;
   709→    const off = new OfflineAudioContext(2, Math.ceil(44100 * dur), 44100);
   710→    const graph = buildGraph(off, project, false);
   711→    // decode user pad samples up-front so they are included in the render
   712→    const padBufs = new Map<DrumSound, AudioBuffer | null>();
   713→    for (const pad of project.pads) {
   714→      if (pad.custom?.dataUrl && !padBufs.has(pad.sound)) {
   715→        padBufs.set(pad.sound, await this.decodeDataUrl(pad.custom.dataUrl));
   716→      }
   717→    }
   718→    const anySolo = project.tracks.some((t) => t.solo);
   719→    let t = 0.05;
   720→    for (let i = 0; i < totalSteps; i++) {
   721→      const step = i % 16;
   722→      const swingOffset = step % 2 === 1 ? project.swing * stepDur * 0.5 : 0;
   723→      const time = t + swingOffset;
   724→      for (const track of project.tracks) {
   725→        const ch = graph.channels.get(track.id);
   726→        if (!ch) continue;
   727→        const audible = !track.muted && (!anySolo || track.solo);
   728→        if (!audible) continue;
   729→        if (track.kind === "drum") {
   730→          const s = track.steps[step];
   731→          if (s?.on) {
   732→            scheduleDrum(off, ch.input, track.sound, time, {
   733→              vel: s.vel,
   734→              tune: track.tune,
   735→              customBuf: padBufs.get(track.sound) ?? null,
   736→            });
   737→          }
   738→        } else {
   739→          for (const n of track.notes) {
   740→            if (n.step === step) {
   741→              scheduleSynth(off, ch.input, track, n.pitch, n.vel, time, n.len * stepDur);
   742→            }
   743→          }
   744→        }
   745→      }
   746→      t += stepDur;
   747→    }
   748→    const rendered = await off.startRendering();
   749→    return encodeWav(rendered);
   750→  }
   751→}

   753→/* ---------------- WAV encoding ---------------- */
   754→
   755→export function encodeWav(buffer: AudioBuffer): Blob {
   756→  const numCh = buffer.numberOfChannels;
   757→  const len = buffer.length;
   758→  const bytes = len * numCh * 2;
   759→  const ab = new ArrayBuffer(44 + bytes);
   760→  const view = new DataView(ab);
   761→  const wstr = (off: number, s: string) => {
   762→    for (let i = 0; i < s.length; i++) view.setUint8(off + i, s.charCodeAt(i));
   763→  };
   764→  wstr(0, "RIFF");
   765→  view.setUint32(4, 36 + bytes, true);
   766→  wstr(8, "WAVE");
   767→  wstr(12, "fmt ");
   768→  view.setUint32(16, 16, true);
   769→  view.setUint16(20, 1, true); // PCM
   770→  view.setUint16(22, numCh, true);
   771→  view.setUint32(24, buffer.sampleRate, true);
   772→  view.setUint32(28, buffer.sampleRate * numCh * 2, true);
   773→  view.setUint16(32, numCh * 2, true);
   774→  view.setUint16(34, 16, true);
   775→  wstr(36, "data");
   776→  view.setUint32(40, bytes, true);
   777→
   778→  const chans: Float32Array[] = [];
   779→  for (let c = 0; c < numCh; c++) chans.push(buffer.getChannelData(c));
   780→  let off = 44;
   781→  for (let i = 0; i < len; i++) {
   782→    for (let c = 0; c < numCh; c++) {
   783→      let s = Math.max(-1, Math.min(1, chans[c][i]));
   784→      s = s < 0 ? s * 0x8000 : s * 0x7fff;
   785→      view.setInt16(off, s, true);
   786→      off += 2;
   787→    }
   788→  }
   789→  return new Blob([ab], { type: "audio/wav" });
   790→}
